# LogTail Drag & Drop Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add comprehensive drag and drop functionality to LogTail, including tab system, file drop from OS, drag log lines out, and visual polish.

**Architecture:** Avalonia Native approach using built-in `TabControl` and `DragDrop` APIs. Each tab wraps a `TabViewModel` with its own file state. Drag operations use `DataFormats.Files` for drops and `DataFormats.Text` for drag-out.

**Tech Stack:** Avalonia 11, ReactiveUI, C# 12

## Global Constraints

- Target: .NET 8.0
- UI Framework: Avalonia 11.x
- MVVM: ReactiveUI (ReactiveObject, ReactiveCommand)
- No external UI dependencies beyond Avalonia
- File filter: `.log` and `.txt` only
- Line number format: `{number}. {message}`

// src/LogTail.UI/ViewModels/MainWindowViewModel.cs (add to existing class)
using System.Collections.ObjectModel;
using System.Reactive;
using System.Reactive.Linq;
using ReactiveUI;

namespace LogTail.UI.ViewModels;

public sealed class MainWindowViewModel : ReactiveObject
{
    // ... existing properties ...
    
    private TabViewModel? _selectedTab;
    
    public TabViewModel? SelectedTab
    {
        get => _selectedTab;
        set => this.RaiseAndSetIfChanged(ref _selectedTab, value);
    }
    
    public ObservableCollection<TabViewModel> Tabs { get; } = new();
    
    // Add to existing OpenFile method or create new
    public void AddTab(string filePath)
    {
        // Check if file already open
        var existingTab = Tabs.FirstOrDefault(t => 
            string.Equals(t.FilePath, filePath, StringComparison.OrdinalIgnoreCase));
        
        if (existingTab != null)
        {
            SelectedTab = existingTab;
            return;
        }
        
        var tab = new TabViewModel(filePath);
        Tabs.Add(tab);
        SelectedTab = tab;
    }
    
    public void CloseTab(TabViewModel tab)
    {
        Tabs.Remove(tab);
        
        if (SelectedTab == tab)
        {
            SelectedTab = Tabs.FirstOrDefault();
        }
    }
    
    public TabViewModel? FindTabByPath(string filePath)
    {
        return Tabs.FirstOrDefault(t => 
            string.Equals(t.FilePath, filePath, StringComparison.OrdinalIgnoreCase));
    }
}

// src/LogTail.UI/Views/MainWindow.axaml.cs
using Avalonia.Controls;
using Avalonia.Input;
using Avalonia.Interactivity;
using LogTail.UI.ViewModels;

namespace LogTail.UI.Views;

public partial class MainWindow : Window
{
    private MainWindowViewModel ViewModel => (MainWindowViewModel)DataContext!;
    
    public MainWindow()
    {
        InitializeComponent();
        
        // DragDrop handlers
        AddHandler(DragDrop.DropEvent, OnDrop);
        AddHandler(DragDrop.DragOverEvent, OnDragOver);
    }
    
    private void OnDragOver(object? sender, DragEventArgs e)
    {
        if (e.Data.Contains(DataFormats.Files))
        {
            e.DragEffects = DragDropEffects.Copy;
        }
        else
        {
            e.DragEffects = DragDropEffects.None;
        }
    }
    
    private void OnDrop(object? sender, DragEventArgs e)
    {
        if (!e.Data.Contains(DataFormats.Files))
            return;
        
        var files = e.Data.GetFiles();
        if (files == null)
            return;
        
        foreach (var file in files)
        {
            var path = file.Path.LocalPath;
            if (IsValidLogFile(path))
            {
                ViewModel.AddTab(path);
            }
        }
    }
    
    private static bool IsValidLogFile(string path)
    {
        if (!System.IO.File.Exists(path))
            return false;
        
        var ext = System.IO.Path.GetExtension(path).ToLowerInvariant();
        return ext is ".log" or ".txt";
    }
}
```

// MainWindow.axaml.cs - Add log line drag handler
private void OnLogLineDrag(object? sender, DragEventArgs e)
{
    if (sender is not ListBox listBox)
        return;
    
    var selectedItems = listBox.SelectedItems;
    if (selectedItems == null || selectedItems.Count == 0)
    {
        e.DragEffects = DragDropEffects.None;
        return;
    }
    
    var text = string.Join(Environment.NewLine,
        selectedItems.Cast<EnrichedLogEvent>().Select(x => x.Raw.Line));
    
    e.Data.Set(DataFormats.Text, text);
    e.DragEffects = DragDropEffects.Copy;
}

public string StatusText

    set => this.RaiseAndSetIfChanged(ref _statusText, value);

partial void OnSelectedTabChanged(TabViewModel? value)
{
    UpdateStatusBar();
}

private void UpdateStatusBar()
{
    if (SelectedTab == null)
    {
        StatusText = "No file open";
        return;
    }
    
    StatusText = $"[{SelectedTab.FileName}] {SelectedTab.FilePath} | " +
                 $"{FormatFileSize(SelectedTab.FileSize)} | " +
                 $"Modified: {SelectedTab.LastModified:yyyy-MM-dd HH:mm} | " +
                 $"{SelectedTab.LinesPerSecond:F0} lines/s | " +
                 $"{(SelectedTab.IsTailing ? "Tailing" : "Paused")}";
}

private static string FormatFileSize(long bytes)
{
    return bytes switch
    {
        < 1024 => $"{bytes} B",
        < 1024 * 1024 => $"{bytes / 1024.0:F1} KB",
        < 1024 * 1024 * 1024 => $"{bytes / (1024.0 * 1024):F1} MB",
        _ => $"{bytes / (1024.0 * 1024 * 1024):F1} GB"
    };
}

// src/LogTail.UI/Views/MainWindow.axaml.cs
private void OnDrop(object? sender, DragEventArgs e)
{
    if (!e.Data.Contains(DataFormats.Files))
        return;
    
    var files = e.Data.GetFiles();
    if (files == null)
        return;
    
    foreach (var file in files)
    {
        var path = file.Path.LocalPath;
        
        if (!System.IO.File.Exists(path))
        {
            ViewModel.StatusText = $"Error: File not found - {path}";
            continue;
        }
        
        if (!IsValidLogFile(path))
        {
            continue; // Silently ignore invalid types
        }
        
        try
        {
            // Test if file is readable
            using var stream = System.IO.File.OpenRead(path);
            ViewModel.AddTab(path);
        }
        catch (UnauthorizedAccessException)
        {
            ViewModel.StatusText = $"Error: Cannot read file - access denied: {path}";
        }
        catch (System.IO.IOException ex)
        {
            ViewModel.StatusText = $"Error: File is locked - {ex.Message}";
        }
    }
}
```
